  const analysisPrompt = Number(task3.module) === 2 ? "Describe the setting of the extract." : "State the writer's purpose of the extract.";

    summaryPrompt,
    wordLimit: 50,
    rubric: Object.freeze({

        "The summary must not exceed 50 words."
      ])
    })
  });
}

    if (source2.has(item2)) copied += 1;
  });
  return copied / candidate.size;
}
function situationCoverage(task3, response) {
  const rows = (task3?.stimulus?.situation || []).map((line) => {
    const terms = tokenise(line).map(stem).filter((w) => w.length >= 4 && !STOPWORDS.has(w));
    const candidate = new Set(tokenise(response).map(stem));
    const matched = terms.filter((w) => candidate.has(w));
    const needed = Math.min(2, Math.max(1, Math.ceil(terms.length * 0.2)));
    return { line, matched: matched.slice(0, 4), covered: matched.length >= needed };
  });
  return { rows, covered: rows.filter((r) => r.covered).length, total: rows.length };
}
function diagnostics(response) {
  const w = tokenise(response);
  const sentences = sentenceList(response);
  const paras = paragraphList(response);
  const avgSentence = sentences.length ? w.length / sentences.length : 0;
  return {
    wordCount: countEnglishWords(response),
    sentenceCount: sentences.length,
    paragraphCount: paras.length,
    avgSentence,
    longSentences: sentences.filter((s) => tokenise(s).length > 35).length,
    veryShort: sentences.filter((s) => tokenise(s).length < 4).length,
    lowercaseStarts: sentences.filter((s) => /^[a-z]/.test(s)).length,
    repeatedPunct: (String(response).match(/[!?]{2,}|\.{4,}/g) || []).length,
    diversity: uniqueRatio(response)
  };
}
function languageBand(response, max = 7) {
  const d = diagnostics(response);
  if (d.wordCount < 20) return { score: 0, notes: ["The response is too short to demonstrate sustained language control."], diagnostics: d };
  let score = max;
  const notes = [];
  if (d.diversity < 0.3) {
    score -= 1;
    notes.push("Vocabulary is repetitive. Vary word choice where it improves precision.");
  }
  if (d.avgSentence < 7 || d.avgSentence > 30) {
    score -= 1;
    notes.push("Sentence length needs more control and variety.");
  }
  if (d.longSentences >= 2) {
    score -= 1;
    notes.push("Several sentences are very long and may need clearer punctuation or division.");
  }
  if (d.veryShort >= Math.max(3, Math.ceil(d.sentenceCount * 0.3))) {
    score -= 0.7;
    notes.push("Frequent very short sentences make the writing choppy.");
  }
  if (d.lowercaseStarts > 0) {
    score -= 0.5;
    notes.push("Check sentence capitalisation.");
  }
  if (d.repeatedPunct > 0) {
    score -= 0.3;
    notes.push("Proofread repeated punctuation.");
  }
  if (!notes.length) notes.push("Language is generally controlled, varied and readable.");
  return { score: round(clamp(score, 0, max)), notes, diagnostics: d };
}
function formatEvidence(task3, response) {
  const markers = FORM_MARKERS[task3?.format] || [];
  const raw = String(response || "");
  const low = clean(raw).toLowerCase();
  const matched = new Set(markers.filter((m) => low.includes(m)));
  const format = String(task3?.format || "").toLowerCase();
  if (format === "report") {
    if (/^\s*report\b/im.test(raw) || /\breport on\b/i.test(raw)) matched.add("report heading");
    if (/^\s*to\s*:/im.test(raw)) matched.add("to:");
    if (/^\s*from\s*:/im.test(raw)) matched.add("from:");
    if (/^\s*date\s*:/im.test(raw)) matched.add("date:");
    if (/^\s*subject\s*:/im.test(raw)) matched.add("subject:");
    if (/^\s*\d+(?:\.\d+)?[.)]?\s+/m.test(raw)) matched.add("numbered sections");
    if (/\bfindings?\b/i.test(raw)) matched.add("findings");
    if (/\brecommendations?\b/i.test(raw)) matched.add("recommendations");
    if (/\bconclusions?\b/i.test(raw)) matched.add("conclusion");
  }
  if (format === "speech") {
    if (/\b(?:good morning|good afternoon|good evening|fellow members|friends|members)\b/i.test(raw)) matched.add("audience greeting");
    if (/\b(?:thank you|i urge you|let us|vote yes|vote no)\b/i.test(raw)) matched.add("speech close");
  }
  return { markers, matched: [...matched] };
}
function wordRangePenalty(task3, response) {
  const wc = countEnglishWords(response);
  if (task3?.wordLimit) {
    if (wc <= task3.wordLimit) return { factor: 1, wordCount: wc };
    const over = (wc - task3.wordLimit) / task3.wordLimit;
    return { factor: clamp(1 - over * 1.6, 0.35, 1), wordCount: wc };
  }
  if (task3?.wordRange) {
    const [min, max] = task3.wordRange;
    if (wc >= min && wc <= max) return { factor: 1, wordCount: wc };
    if (wc < min) return { factor: clamp(wc / min, 0.35, 1), wordCount: wc };
    return { factor: clamp(1 - (wc - max) / Math.max(100, max), 0.6, 1), wordCount: wc };
  }
  return { factor: 1, wordCount: wc };
}
function gradeAnalysisPart(task3, response) {
  const low = clean(response).toLowerCase();
  const coverage = sourceCoverage(task3?.stimulus, response);
  if (!low) return { score: 0, evidence: [], feedback: "Part (a) was not answered." };
  let typeSignal = 0;
  if (Number(task3.module) === 2) {
    const settingSignals = ["in ", "at ", "during", "morning", "afternoon", "evening", "night", "day", "school", "home", "town", "village", "road", "room", "centre", "center"];
    typeSignal = Math.min(1.5, countMatches(" " + low, settingSignals) * 0.75);
  } else if (Number(task3.module) === 3) {
    typeSignal = /\b(persuade|convince|argue|urge|encourage|warn|discourage)\b/.test(low) ? 1.5 : 0.5;
  } else {
    typeSignal = /\b(inform|explain|describe|show|outline|educate|identify)\b/.test(low) ? 1.5 : 0.5;
  }
  const relevance = Math.min(1.5, coverage.matched.length * 0.5);
  return {
    score: round(clamp(typeSignal + relevance, 0, 3)),
    evidence: coverage.matched.slice(0, 5),
    feedback: Number(task3.module) === 2 ? "A strong answer identifies where/when the extract occurs and any important surrounding conditions." : "A strong answer states what the writer is trying to achieve and identifies the subject of that purpose."
  };
}
function gradeSummary(task3, response, analysisResponse) {
  const analysis = gradeAnalysisPart(task3, analysisResponse);
  const coverage = sourceCoverage(task3?.stimulus, response);
  const ideas = sourceIdeaCoverage(task3?.stimulus, response);
  const contradictions = contradictorySourceSignals(task3?.stimulus, response);
  const lang = languageBand(response, 4);
  const copyRatio = copiedPhraseRatio(task3?.stimulus, response);
  const length = wordRangePenalty(task3, response);
  let understanding = ideas.total ? Math.min(3, ideas.covered) : round(clamp(coverage.ratio * 4.5, 0, 3));
  if (countEnglishWords(response) < 15) understanding = Math.min(understanding, 1.5);
  if (contradictions.count > 0) understanding = Math.min(understanding, 2);
  let evaluating = lang.score;
  if (copyRatio > 0.35) evaluating = Math.min(evaluating, 2);
  else if (copyRatio > 0.2) evaluating = Math.min(evaluating, 3);
  if (length.factor < 1) evaluating = round(evaluating * length.factor);
  const score = round(clamp(analysis.score + understanding + evaluating, 0, 10));
  const feedback = [
    analysis.feedback,
    ideas.total ? `Part (b) clearly represents ${ideas.covered} of ${ideas.total} distinct source idea blocks.` : `Part (b) covers ${coverage.matched.length} of ${coverage.terms.length} high-value source terms detected by SPARK.`,
    `Summary word count: ${length.wordCount}/${task3.wordLimit || 50}.`
  ];
  if (copyRatio > 0.2) feedback.push("Several phrases closely match the source. Paraphrase more of the summary in your own words.");
  if (contradictions.count > 0) feedback.push("SPARK detected negation around source ideas, so content credit was capped rather than assuming those ideas were stated correctly.");
  feedback.push(...lang.notes);
  return {
    score,
    maxMarks: 10,
    confidence: "medium-high",
    dimensions: [
      { id: "analysing", label: "P2 Analysing: Part (a)", score: analysis.score, max: 3, evidence: analysis.evidence },
      { id: "understanding", label: "P1 Understanding: THREE distinct summary points", score: understanding, max: 3, evidence: ideas.total ? ideas.ideas.filter((row) => row.covered) : coverage.matched.slice(0, 6) },
      { id: "evaluating", label: "P3 Evaluating & Creating: language and mechanics", score: evaluating, max: 4, evidence: lang.notes }
    ],
    feedback,
    diagnostics: { ...lang.diagnostics, copyRatio: round(copyRatio), distinctSourceIdeas: ideas.covered, contradictionSignals: contradictions.count }
  };
}
function gradeExposition(task3, response) {
  const cov = situationCoverage(task3, response);
  const lang = languageBand(response, 8);
  const fmt = formatEvidence(task3, response);
  const d = lang.diagnostics;
  const connectors = countMatches(response, CONNECTORS);
  const length = wordRangePenalty(task3, response);
  const relevance = cov?.total ? Math.min(3, cov.covered / cov.total * 3) : 1.5;
  const completeness = cov?.total ? Math.min(4, cov.covered / cov.total * 4) : 2;
  const understanding = round(relevance + completeness);
  const formatName = String(task3?.format || "").toLowerCase();
  const formatMarks = formatName === "report" ? fmt.matched.length >= 7 ? 3 : fmt.matched.length >= 4 ? 2 : fmt.matched.length >= 2 ? 1 : 0 : fmt.markers.length ? Math.min(3, fmt.matched.length / Math.min(2, fmt.markers.length) * 3) : 2;
  const audience = /\b(you|your|we|our|principal|manager|students|council|community|readers)\b/i.test(response) ? 2 : 1;
  const structuredReport = formatName === "report" && fmt.matched.includes("numbered sections") && fmt.matched.includes("recommendations");
  const sequencing = d.paragraphCount >= 3 && connectors >= 2 || structuredReport ? 2 : d.paragraphCount >= 2 ? 1 : 0;
  const analysing = round(clamp(formatMarks + audience + sequencing, 0, 7));
  const concision = round(clamp(3 * length.factor, 0, 3));
  const coherence = connectors >= 3 && d.paragraphCount >= 3 ? 2 : connectors >= 1 ? 1 : 0;
  const mechanics = round(clamp(3 - (d.lowercaseStarts * 0.4 + d.repeatedPunct * 0.4 + d.longSentences * 0.2), 0, 3));
  const language = round(clamp(lang.score * length.factor, 0, 8));
  const evaluating = round(clamp(concision + coherence + language + mechanics, 0, 16));
  return {
    score: round(understanding + analysing + evaluating),
    maxMarks: 30,
    confidence: cov?.total ? "medium-high" : "medium",
    dimensions: [
      { id: "understanding", label: "P1 Understanding: relevance, clarity, accuracy and completeness", score: understanding, max: 7, evidence: cov?.rows || [] },
      { id: "analysing", label: "P2 Analysing: format, audience and sequencing", score: analysing, max: 7, evidence: fmt.matched },
      { id: "evaluating", label: "P3 Evaluating & Creating: concision, coherence, language and mechanics", score: evaluating, max: 16, evidence: lang.notes }
    ],
    feedback: [
      cov?.total ? `${cov.covered}/${cov.total} stimulus points are clearly represented.` : "Check that every required stimulus point is used.",
      `Format markers detected: ${fmt.matched.length ? fmt.matched.join(", ") : "none"}.`,
      `Word count: ${d.wordCount}.`,
      ...lang.notes
    ],
    diagnostics: d
  };
}
function gradeLiterary(task3, response) {
  const lang = languageBand(response, 7);
  const d = lang.diagnostics;
  const low = clean(response).toLowerCase();
  const dialogue = (String(response).match(/[“"][^”"]+[”"]/g) || []).length;
  const narrative = countMatches(response, NARRATIVE);
  const sensory = countMatches(response, SENSORY);
  const resolution = countMatches(response, RESOLUTION);
  const promptTerms = tokenise(task3.instructions).map(stem).filter((w) => w.length >= 5 && !STOPWORDS.has(w)).slice(-12);
  const candidate = new Set(tokenise(response).map(stem));
  const promptHits = promptTerms.filter((w) => candidate.has(w)).length;
  const length = wordRangePenalty(task3, response);
  const promptUse = Math.min(4, 1 + promptHits * 0.8);
  const storyElements = Math.min(6, (narrative >= 3 ? 2 : 1) + (resolution ? 1.5 : 0.5) + (dialogue ? 1.25 : 0.5) + (sensory >= 2 ? 1.25 : 0.5));
  const development = Math.min(6, d.wordCount >= 300 ? 6 : d.wordCount >= 200 ? 4 : d.wordCount >= 120 ? 2 : 1);
  const evaluating = round(clamp((promptUse + storyElements + development) * length.factor, 0, 16));
  const analysing = round(clamp((d.paragraphCount >= 5 ? 3 : d.paragraphCount >= 3 ? 2 : 1) + (narrative >= 4 ? 2 : narrative >= 2 ? 1 : 0) + (resolution ? 2 : 1), 0, 7));
  const understanding = round(clamp(lang.score, 0, 7));
  const score = round(clamp(evaluating + analysing + understanding, 0, 30));
  const feedback = [
    promptHits ? "The response is visibly connected to the chosen stimulus." : "Make the required stimulus unmistakably central to the story.",
    dialogue ? `Dialogue appears ${dialogue} time(s); make sure it develops character, action or atmosphere.` : "Dialogue is optional, but purposeful dialogue can help characterisation.",
    sensory >= 2 ? "Sensory detail is used to build the scene." : "Use selective sensory detail to establish setting and atmosphere.",
    resolution ? "A resolution/change signal is present." : "Make the conflict and ending feel causally connected rather than abrupt.",
    d.wordCount < 200 ? "The specimen mark scheme caps very short stories (under about 200 words) at a low range. Develop the story further." : `Word count: ${d.wordCount}.`,
    ...lang.notes
  ];
  return {
    score,
    maxMarks: 30,
    confidence: "medium",
    dimensions: [
      { id: "evaluating", label: "P3 Evaluating & Creating: content and relevance of story", score: evaluating, max: 16 },
      { id: "analysing", label: "P2 Analysing: organisation and sequencing", score: analysing, max: 7 },
      { id: "understanding", label: "P1 Understanding: language, grammar, mechanics and narrative control", score: understanding, max: 7, evidence: lang.notes }
    ],
    feedback,
    diagnostics: d
  };
}
function persuasiveSignals(response) {
  const raw = String(response || "");
  const low = clean(raw).toLowerCase();
  const paragraphs = paragraphList(raw);
  const rhetorical = (raw.match(/\?/g) || []).length;
  const directAddress = (low.match(/\b(?:you|your|we|our|us|fellow members|friends|members|community)\b/g) || []).length;
  const calls = (low.match(/\b(?:i urge you|let us|join|support|reject|vote yes|vote no|act now|we must|we should)\b/g) || []).length;
  const repetition = (low.match(/\bthink of\b/g) || []).length;
  const counterOpeners = (low.match(/\b(?:some will say|some may say|some may argue|some people argue|others say|critics may argue|however|although|admittedly|on the other hand)\b/g) || []).length;
  const rebuttals = (low.match(/\b(?:i disagree|but|however|yet|that concern|this concern|in response|the answer is|does not|do not)\b/g) || []).length;
  const supportSignals = (low.match(/\b(?:because|for example|for instance|such as|think of|consider|according to|survey|data|evidence|cost|benefit|help|helps|allow|allows|give|gives|provide|provides|means|therefore|so that|as a result|in an emergency)\b/g) || []).length;
  const stance = /\b(?:i believe|i think|i support|i oppose|i stand firmly|in favour|in favor|against|should|should not|must|must not|in my view|my position)\b/.test(low);
  const hasOpening = paragraphs.slice(0, 2).some((p) => /\b(?:good morning|good afternoon|good evening|fellow members|friends|members|today|i stand|i believe|i support|i oppose|in favour|in favor|against)\b/i.test(p));
  const hasConclusion = paragraphs.slice(-2).some((p) => /\b(?:therefore|so let us|let us|i urge|in conclusion|finally|vote|thank you|must act|should support|should reject)\b/i.test(p));
  return { paragraphs, rhetorical, directAddress, calls, repetition, counterOpeners, rebuttals, supportSignals, stance, hasOpening, hasConclusion };
}
function gradePersuasive(task3, response) {
  const lang = languageBand(response, 7);
  const d = lang.diagnostics;
  const sig = persuasiveSignals(response);
  const length = wordRangePenalty(task3, response);
  const position = sig.stance ? 3 : 1;
  const support = Math.min(5, sig.supportSignals >= 8 ? 5 : sig.supportSignals >= 5 ? 4 : sig.supportSignals >= 3 ? 3 : sig.supportSignals >= 1 ? 1.5 : 0);
  const audience = Math.min(2, sig.directAddress >= 4 ? 2 : sig.directAddress ? 1.5 : 0.5);
  const opposition = Math.min(2, sig.counterOpeners && sig.rebuttals ? 2 : sig.counterOpeners ? 1 : 0);
  const devices = Math.min(4, (sig.rhetorical ? 1 : 0) + (sig.repetition >= 2 ? 1 : sig.repetition ? 0.5 : 0) + (sig.calls ? 1 : 0) + (sig.directAddress >= 4 ? 1 : 0));
  let evaluating = round(clamp((position + support + audience + opposition + devices) * length.factor, 0, 16));
  const structure = sig.paragraphs.length >= 5 ? 2 : sig.paragraphs.length >= 3 ? 1 : 0;
  const progression = sig.supportSignals >= 5 ? 2 : sig.supportSignals >= 2 ? 1 : 0;
  const counterFlow = sig.counterOpeners && sig.rebuttals ? 1 : 0;
  const framing = (sig.hasOpening ? 1 : 0) + (sig.hasConclusion ? 1 : 0);
  let analysing = round(clamp(structure + progression + counterFlow + framing, 0, 7));
  let understanding = round(clamp(lang.score, 0, 7));
  if (d.wordCount < 150) {
    evaluating = Math.min(evaluating, 6);
    analysing = Math.min(analysing, 2);
    understanding = Math.min(understanding, 2);
  }
  const score = round(clamp(evaluating + analysing + understanding, 0, 30));
  const deviceNames = [sig.rhetorical ? "rhetorical question" : "", sig.repetition ? "repetition" : "", sig.calls ? "call to action" : "", sig.directAddress >= 4 ? "direct audience address" : ""].filter(Boolean);
  return {
    score,
    maxMarks: 30,
    confidence: "medium-high",
    dimensions: [
      { id: "evaluating", label: "P3 Evaluating & Creating: position, support, audience and persuasive technique", score: evaluating, max: 16 },
      { id: "analysing", label: "P2 Analysing: argument organisation and logical sequence", score: analysing, max: 7 },
      { id: "understanding", label: "P1 Understanding: language, grammar, mechanics and persuasive control", score: understanding, max: 7, evidence: lang.notes }
    ],
    feedback: [
      sig.stance ? "A clear position is established and maintained." : "State a clear position early and maintain it.",
      sig.counterOpeners && sig.rebuttals ? "The response raises an opposing view and answers it." : "A relevant opposing view and rebuttal can strengthen the argument.",
      sig.supportSignals ? "The response develops reasons with examples, effects or supporting explanation." : "Develop reasons with examples, evidence, causes/effects or comparisons.",
      deviceNames.length ? "Persuasive devices detected: " + deviceNames.join(", ") + "." : "Use purposeful persuasive devices rather than relying only on assertion.",
      sig.hasOpening && sig.hasConclusion ? "The argument has a recognisable opening and conclusion." : "Strengthen the introduction and conclusion so the argument is clearly framed.",
      d.paragraphCount >= 5 ? "The response has a developed multi-paragraph structure." : "The specimen mark scheme expects a developed structure with introduction, body paragraphs and conclusion.",
      d.wordCount < 150 ? "The specimen mark scheme states that a response under about 150 words cannot earn above the low range." : "Word count: " + d.wordCount + ".",
      ...lang.notes
    ],
    diagnostics: { ...d, persuasiveSignals: { ...sig, paragraphs: sig.paragraphs.length } }
  };
}
function gradeEnglishAPaper2Response(task3, response, analysisResponse = "") {
  const value2 = String(response || "");
  if (task3?.kind === "summary") {
    if (!value2.trim() && !String(analysisResponse || "").trim()) return { score: 0, maxMarks: 10, confidence: "high", dimensions: [], feedback: ["No response was submitted."], diagnostics: diagnostics(value2) };
    return gradeSummary(task3, value2, analysisResponse);
  }
  if (!value2.trim()) return { score: 0, maxMarks: task3?.rubric?.marks || 0, confidence: "high", dimensions: [], feedback: ["No response was submitted."], diagnostics: diagnostics(value2) };
  if (task3.kind === "exposition") return gradeExposition(task3, value2);
  if (task3.kind === "persuasive") return gradePersuasive(task3, value2);
  if (task3.kind === "literary") return gradeLiterary(task3, value2);
  return { score: 0, maxMarks: task3?.rubric?.marks || 0, confidence: "low", dimensions: [], feedback: ["This task type does not yet have a practice rubric."], diagnostics: diagnostics(value2) };
}
function gradeEnglishAPaper2(paper, answers = {}, choiceId = "") {
  const allTasks = paper?.tasks || [];
  const normalTasks = allTasks.filter((task3) => !task3.choiceGroup);
  const choiceGroups = [...new Set(allTasks.map((task3) => task3.choiceGroup).filter(Boolean))];
  const choiceTasks = choiceGroups.map((group) => {
    const options = allTasks.filter((task3) => task3.choiceGroup === group);
    return options.find((task3) => task3.id === choiceId) || options[0];
  }).filter(Boolean);
  const tasks = [...normalTasks, ...choiceTasks];
  const rows = tasks.map((task3) => ({
    task: task3,
    ...gradeEnglishAPaper2Response(
      task3,
      task3.choiceGroup && task3.id !== choiceId ? "" : answers[task3.id] || "",
      task3.choiceGroup && task3.id !== choiceId ? "" : answers[`${task3.id}:analysis`] || ""
    )
  }));
  const score = round(rows.reduce((sum, row) => sum + row.score, 0));
  const maxScore = rows.reduce((sum, row) => sum + row.maxMarks, 0);
  return {
    rows,
    score,
    maxScore,
    percent: maxScore ? Math.round(score / maxScore * 100) : 0,
    modules: [1, 2, 3].map((module) => {
      const moduleRows = rows.filter((row) => row.task.module === module);
      return { module, score: round(moduleRows.reduce((sum, row) => sum + row.score, 0)), max: moduleRows.reduce((sum, row) => sum + row.maxMarks, 0) };
    }),
    note: "SPARK practice estimate aligned to the profile structure in the CXC 01/G/SYLL 25 specimen mark scheme. It checks task fulfilment, stimulus use, organisation, language and mechanics and shows the evidence behind each mark. Extended-writing marks are automated practice estimates and may differ from official examination marking."
  };
}

var CONSTANTS = { pi: Math.PI, PI: Math.PI, e: Math.E };
var FUNCTIONS = {
  sqrt: Math.sqrt,
  abs: Math.abs,
  sin: null,
  cos: null,
  tan: null,
  asin: null,
  acos: null,
  atan: null,
  log: Math.log10,
  ln: Math.log
};
var DEG = Math.PI / 180;
FUNCTIONS.sin = (x) => Math.sin(x * DEG);
FUNCTIONS.cos = (x) => Math.cos(x * DEG);
FUNCTIONS.tan = (x) => Math.tan(x * DEG);
FUNCTIONS.asin = (x) => Math.asin(x) / DEG;
FUNCTIONS.acos = (x) => Math.acos(x) / DEG;
FUNCTIONS.atan = (x) => Math.atan(x) / DEG;
FUNCTIONS.bearing = (north, east) => (Math.atan2(east, north) / DEG + 360) % 360;
function normalise(raw) {
  let s = String(raw ?? "").trim();
  if (!s) return "";
  s = s.replace(/(?<![\w.])(\d+(?:\.\d*)?|\.\d+)[eE]([+-]?\d+)(?![\w.])/g, "($1*10^($2))").replace(/[−–—]/g, "-").replace(/[×∗]/g, "*").replace(/[÷]/g, "/").replace(/√/g, "sqrt").replace(/π/g, "pi").replace(/≤/g, "<=").replace(/≥/g, ">=").replace(/°/g, "").replace(/\\vec\{([^}]*)\}/g, "$1").replace(/[,\s]+/g, (m) => /,/.test(m) ? "," : " ").replace(/\s*\^\s*/g, "^");
  s = s.replace(/(\d)\s*sqrt/g, "$1*sqrt");
  s = s.replace(
    /(^|[\s(=+\-*/])(\d+)\s+(\d+)\s*\/\s*(\d+)/g,
    (_, pre, w, n, d) => `${pre}(${w}+${n}/${d})`
  );
  s = s.replace(
    /\b(sqrt|sin|cos|tan|asin|acos|atan|log|ln|abs)\s*(-?\d+(?:\.\d+)?|[A-Za-z])(?![\w(])/g,
    (_, fn, arg) => `${fn}(${arg})`
  );
  return s;
}
var NAMES = Object.keys(FUNCTIONS).concat(Object.keys(CONSTANTS)).filter((n) => n.length > 1).sort((a, b) => b.length - a.length);
function splitWord(word, out) {
  let rest = word;
  while (rest.length) {
    const name = NAMES.find((n) => rest.startsWith(n));
    if (name) {
      out.push({ t: name in FUNCTIONS ? "fn" : "var", v: name });
      rest = rest.slice(name.length);
    } else {
      out.push({ t: "var", v: rest[0] });
      rest = rest.slice(1);
    }
  }
}
function tokenise2(s) {
  const out = [];
  let i = 0;
  while (i < s.length) {
    const c = s[i];
    if (c === " ") {
      i += 1;
      continue;
    }
    if (/[0-9.]/.test(c)) {
      let j = i;
      while (j < s.length && /[0-9.]/.test(s[j])) j += 1;
      out.push({ t: "num", v: s.slice(i, j).replace(/^0+(?=\d)/, "") });
      i = j;
      continue;
    }

      continue;
    }

      i += 2;
      continue;
    }

  }

}
function explicitMultiplication(s) {
  const tk = tokenise2(s);
  const out = [];
  for (let i = 0; i < tk.length; i += 1) {
    const a = tk[i];
    const b = tk[i + 1];
    out.push(a.v);
    if (!b) break;
    const endsValue = a.t === "num" || a.t === "var" || a.t === ")";
    const startsValue = b.t === "num" || b.t === "var" || b.t === "fn" || b.t === "(";
    if (endsValue && startsValue) out.push("*");
  }
  return out.join("");
}

    if (eq.length === 2) s = eq[1];
    else if (eq.length > 2) throw new Error("too many = signs");
    s = explicitMultiplication(s).replace(/\^/g, "**");
    if (/[^0-9A-Za-z_.+\-*/()><=!, ]/.test(s)) throw new Error("bad character");
    const names = [...new Set(s.match(/[A-Za-z_][A-Za-z0-9_]*/g) || [])];
    const vars = names.filter((n) => !(n in FUNCTIONS) && !(n in CONSTANTS));
    const args = vars.join(",");
    const body = new Function(
      "__f",
      "__c",
      args,
      `"use strict";const{${Object.keys(FUNCTIONS).join(",")}}=__f;const{${Object.keys(CONSTANTS).join(",")}}=__c;return (${s});`
    );

  } catch {

function readScienceGraph(value2) {
  try {
    const graph = typeof value2 === "string" ? JSON.parse(value2) : value2;
    if (graph?.version !== 1 || !Array.isArray(graph.series) || graph.series.length < 1 || graph.series.length > 3) return null;

  } catch {

function graphPoints(text, kind) {
  return String(text || "").split(/\n|;/).filter((s) => s.trim()).slice(0, 101).map((line) => {
    const match = line.trim().match(/^\(?\s*(.+?)\s*,\s*([+-]?(?:\d+(?:\.\d*)?|\.\d+)(?:[eE][+-]?\d+)?)\s*\)?$/);
    if (!match) return null;
    const x = kind === "bar" ? match[1].trim() : Number(match[1]);
    return (kind === "bar" ? Boolean(x) : Number.isFinite(x)) ? { x, y: Number(match[2]) } : null;
  });
}
function graphAxis(axis) {
  if (!axis || [axis.min, axis.max, axis.step].some((v) => v === "" || v === null || v === void 0)) return null;
  const min = Number(axis.min), max = Number(axis.max), step = Number(axis.step);
  const intervals = (max - min) / step;
  if (![min, max, step].every(Number.isFinite) || max <= min || step <= 0 || intervals < 1 || intervals > 20 || Math.abs(intervals - Math.round(intervals)) > 1e-7) return null;
  return { min, max, step, intervals: Math.round(intervals) };
}

function parseStoredGraphPoints(value2) {
  return String(value2 || "").split(/\n|;/).map((line) => {
    const match = line.match(/^\s*([AB])?\s*:?\s*(-?\d+(?:\.\d+)?)\s*[, ]\s*(-?\d+(?:\.\d+)?)/i);
    return match ? { series: (match[1] || "A").toUpperCase(), x: Number(match[2]), y: Number(match[3]) } : null;
  }).filter((point) => point && Number.isFinite(point.x) && Number.isFinite(point.y));
}
function expectedGraphCoordinates(item2) {
  const text = (item2?.markScheme?.points || []).join(" ");
  const out = [];
  const matcher = /\((-?\d+(?:\.\d+)?)\s*,\s*([+-]?\d+(?:\.\d+)?)\)/g;
  let match;
  while (match = matcher.exec(text)) {
    const point = { x: Number(match[1]), y: Number(match[2]) };
    if (Number.isFinite(point.x) && Number.isFinite(point.y) && !out.some((existing) => existing.x === point.x && existing.y === point.y)) out.push(point);
  }

}
function graphPointMatches(student, target, expected) {
  const xs = expected.map((point) => point.x), ys = expected.map((point) => point.y);
  const xRange = Math.max(1, Math.max(...xs) - Math.min(...xs));
  const yRange = Math.max(1, Math.max(...ys) - Math.min(...ys));
  return Math.abs(student.x - target.x) <= Math.max(0.08, xRange * 0.025) && Math.abs(student.y - target.y) <= Math.max(0.08, yRange * 0.025);
}
function labelMatches(value2, expected) {
  if (!String(value2 || "").trim()) return false;

  if (!expectedTokens.length) return true;

  return hits >= Math.max(1, Math.ceil(expectedTokens.length * 0.6));
}
function graphScaleAssessment(base, responses, expected) {
  const xMin = Number(responses[`${base}:xMin`] ?? 0), xMax = Number(responses[`${base}:xMax`] ?? 10);
  const yMin = Number(responses[`${base}:yMin`] ?? 0), yMax = Number(responses[`${base}:yMax`] ?? 10);
  const valid = [xMin, xMax, yMin, yMax].every(Number.isFinite) && xMax > xMin && yMax > yMin;
  if (!valid) return { valid: false, suitable: false, xMin, xMax, yMin, yMax };
  if (!expected.length) return { valid: true, suitable: true, xMin, xMax, yMin, yMax };
  const xs = expected.map((point) => point.x), ys = expected.map((point) => point.y);
  const allInside = Math.min(...xs) >= xMin && Math.max(...xs) <= xMax && Math.min(...ys) >= yMin && Math.max(...ys) <= yMax;
  const xCoverage = (Math.max(...xs) - Math.min(...xs)) / Math.max(1e-9, xMax - xMin);
  const yCoverage = (Math.max(...ys) - Math.min(...ys)) / Math.max(1e-9, yMax - yMin);
  const usefulCoverage = Math.max(xCoverage, yCoverage) >= 0.6 && Math.min(xCoverage, yCoverage) >= 0.35;
  return { valid: true, suitable: allInside && usefulCoverage, xMin, xMax, yMin, yMax, xCoverage, yCoverage };
}
