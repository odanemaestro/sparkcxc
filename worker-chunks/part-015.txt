async function gradePobArchiveExaminerPaper({
  pack,
  model,
  apiKey,

    throw new Error("historical_examiner_pack_required");
  const parts = pack.parts, questions = [];

    });

      maxPossibleMarks: blank ? 0 : rows.reduce((n, c) => n + c.maxPossibleMarks, 0),

      }))
    };
  });

  return {
    criteria: allCriteria,
    questions,
    parts: partBreakdown,
    score,
    minScore: questions.reduce((n, q) => n + q.minMarks, 0),
    maxScore: questions.reduce((n, q) => n + q.maxPossibleMarks, 0),
    uncertain
  };
}

    (item2) => item2.constraints.questionNumber === number
  ));
  if (chunks.flat().length !== pack.items.length) throw new Error("question_mapping_incomplete");
  const analyses = await Promise.all(chunks.map(async (items) => {
    if (!items.length) return { criteria: [], uncertain: false, score: 0, minScore: 0, maxScore: 0 };
    const [first, second] = await Promise.all([1, 2].map((pass) => requestPass({ model, apiKey, items, responses: pack.responses, pass })));
    return reconcileSemanticPasses(first, second, items, pack.responses);
  }));
  const criteria = analyses.flatMap((result) => result.criteria);
  const semantic = {
    criteria,
    uncertain: analyses.some((result) => result.uncertain),
    score: analyses.some((result) => result.score === null) ? null : analyses.reduce((sum, result) => sum + result.score, 0),
    minScore: analyses.reduce((sum, result) => sum + result.minScore, 0),
    maxScore: analyses.reduce((sum, result) => sum + result.maxScore, 0)
  };
  const breakdown = summarizePobJan2026Marks(semantic, pack);
  if (semantic.score !== breakdown.score || semantic.minScore !== breakdown.minScore || semantic.maxScore !== breakdown.maxScore) throw new Error("examiner_score_mismatch");
  return { ...semantic, parts: breakdown.parts, questions: breakdown.questions };
}

// src/grading/serverAssessment.js
function buildTrustedAssessment(attempt, authority = {}) {
  if (attempt?.status !== "submitted" || attempt.paper !== "02") throw new Error("submitted_paper_required");
  const responses = attempt.response_snapshot;
  if (!responses || typeof responses !== "object" || Array.isArray(responses)) throw new Error("invalid_snapshot");
  if (attempt.subject_id === "principles-of-business" && attempt.bank_version === POB_2026_JAN_HISTORICAL_BANK_VERSION) {
    if (authority.historicalJan2026Enabled !== true) throw new Error("unsupported_bank_version");
    const paper = buildPob2026JanHistoricalPaper();
    if (attempt.metadata?.paper_id !== paper.id || attempt.metadata?.source_sha256 !== paper.sourceSha256 || attempt.score != null || Number(attempt.max_score) !== 100 || attempt.rubric_version !== "pob-historical-2026-jan-p2-rubric-v1" || attempt.grader_version !== SEMANTIC_MARKING_REVISION)
      throw new Error("historical_attempt_identity_mismatch");
    const allowed2 = new Set(paper.questions.flatMap((q) => q.items.map((p) => q.id + ":" + p.id)));
    if (Object.keys(responses).some((id) => !allowed2.has(id)))
      throw new Error("unknown_historical_part");
    const pack = buildPobJan2026ExaminerPack({ responses });
    if (pack.paperId !== paper.id || pack.sourceSha256 !== paper.sourceSha256 || pack.maxMarks !== 100 || pack.items.some((i) => !allowed2.has(i.constraints.primaryResponseKey)))
      throw new Error("historical_source_not_verified");
    return {
      items: pack.items,
      responses: pack.responses,
      deterministicScore: 0,
      maxMarks: 100,
      unassessedMarks: 0,
      baselineScore: 0,
      historicalPaperId: paper.id,
      sourceSha256: paper.sourceSha256,
      parts: pack.parts,
      criterionLevel: true
    };
  }
  if (attempt.subject_id === "principles-of-business" && String(attempt.bank_version || "").startsWith("pob-archive-v1-")) {
    if (authority.historicalPobArchiveEnabled !== true)
      throw new Error("unsupported_bank_version");
    const paperId = attempt.metadata?.paper_id;
    const record = eligiblePobArchiveScoringSources().find((p) => p.id === paperId);
    if (!record || attempt.bank_version !== pobArchiveBankVersion(paperId) || attempt.metadata?.source_sha256 !== record.source.sourceSha256 || attempt.rubric_version !== POB_ARCHIVE_RUBRIC_VERSION || attempt.grader_version !== SEMANTIC_MARKING_REVISION || attempt.score != null || Number(attempt.max_score) !== 100)
      throw new Error("historical_attempt_identity_mismatch");
    const pack = buildPobArchiveExaminerPack({
      paperId,
      selectedQuestions: attempt.metadata?.selected_questions,
      responses
    });
    return { ...pack, deterministicScore: 0, baselineScore: 0, unassessedMarks: 0 };
  }
  const items = [];
  let baseline, replaceScore = 0;
  if (attempt.subject_id === "integrated-science") {
    if (attempt.bank_version !== "integrated-science-v1.2.0") throw new Error("unsupported_bank_version");
    const ids = attempt.metadata?.question_ids;
    if (!Array.isArray(ids) || ids.length !== 6 || new Set(ids).size !== 6) throw new Error("invalid_question_selection");
    const bank = [is_module1_default, is_module2_default, is_module3_default].flatMap((m) => m.paper02);
    const paper = ids.map((id) => bank.find((q) => q.id === id));
    if (paper.some((q) => !q)) throw new Error("unknown_question");
    for (const module of [1, 2, 3]) {
      const selected = paper.filter((q) => q.id.startsWith(`IS-M${module}-`));
      if (selected.length !== 2 || selected.map((q) => Number(q.totalMarks)).sort((a, b) => a - b).join(",") !== "15,20") throw new Error("invalid_module_selection");
    }
    baseline = gradeIntegratedSciencePaper2(paper, responses);
    for (const q of baseline.questions) for (const row of q.items) {
      const part2 = q.question.parts[row.partIndex], item2 = part2.items[row.itemIndex], id = `${q.question.id}:${row.partIndex}:${row.itemIndex}`;
      if ((item2.response?.type || "lines") !== "lines" || !String(responses[id] || "").trim()) continue;
      items.push({
        id,
        maxMarks: row.maxMarks,
        responseKeys: [id],
        prompt: item2.prompt,
        context: [q.question.context, part2.context].filter(Boolean).join("\n"),
        rubric: item2.markScheme
      });
      replaceScore += row.score;
    }
  } else if (attempt.subject_id === "english-a") {
    if (attempt.bank_version !== "english-a-paper2-v1") throw new Error("unsupported_bank_version");
    const paperId = attempt.metadata?.paper_id, choice = attempt.metadata?.selected_creative_prompt || "";
    if (!englishAPaper2Sets.some((p) => p.id === paperId)) throw new Error("unknown_paper");
    const paper = buildEnglishAPaper2(paperId);
    if (choice && !paper.tasks.some((t) => t.choiceGroup && t.id === choice)) throw new Error("invalid_creative_selection");
    baseline = gradeEnglishAPaper2(paper, responses, choice);
    for (const row of baseline.rows) {
      const task3 = row.task;
      if (task3.choiceGroup && task3.id !== choice) continue;
      const keys = task3.kind === "summary" ? [task3.id, `${task3.id}:analysis`] : [task3.id];
      if (!keys.some((key) => String(responses[key] || "").trim())) continue;
      const words = String(responses[task3.id] || "").trim().split(/\s+/).filter(Boolean).length;
      const short = task3.kind === "persuasive" && words < 150 || task3.kind === "literary" && words < 200;
      items.push({
        id: task3.id,
        maxMarks: short ? Math.min(10, row.maxMarks) : row.maxMarks,
        responseKeys: keys,
        prompt: task3.instructions,
        context: task3.stimulus,
        rubric: task3.rubric,
        constraints: { kind: task3.kind, wordLimit: task3.wordLimit, wordRange: task3.wordRange, shortResponseCap: short ? 10 : null }
      });
      replaceScore += row.score;
    }
  } else if (attempt.subject_id === "principles-of-business") {
    if (attempt.bank_version !== "pob-original-paper2-01") throw new Error("unsupported_bank_version");
    if (attempt.metadata?.paper_id !== POB_PAPER2_ORIGINAL_01.id) throw new Error("unknown_paper");
    let determined = 0;
    for (const question2 of POB_PAPER2_ORIGINAL_01.questions) {
      for (const part2 of question2.items) {
        const id = question2.id + ":" + part2.id;
        const value2 = String(responses[id] || "").trim();
        if (!value2) continue;
        const listing = gradePobListingPart(id, value2);
        if (listing) {
          determined += listing.marks;
          continue;
        }
        const numeric = gradePobCalculationResultPart(id, value2);
        const hasVerifiedResult = numeric?.assessedCriterionMarks === 1;
        if (hasVerifiedResult) determined += 1;
        const remaining = part2.criteria.filter((_, index) => !hasVerifiedResult || index !== numeric.criterionIndex);
        const semanticMax = remaining.reduce((sum, c) => sum + c.marks, 0);
        if (semanticMax > 0) items.push({
          id,
          maxMarks: semanticMax,
          responseKeys: [id],
          prompt: part2.prompt,
          context: question2.context,
          rubric: remaining.map((c) => ({ marks: c.marks, rule: c.rule, alternatives: c.alternatives || [] })),
          constraints: {
            subject: "principles-of-business",
            profile: part2.profile,
            deterministicResultMark: hasVerifiedResult ? 1 : 0
          }
        });
      }
    }
    baseline = { score: determined, maxScore: 100, unassessedMarks: 0 };
  } else throw new Error("unsupported_subject");
  const allowed = new Set(items.flatMap((item2) => item2.responseKeys));
  const semanticResponses = Object.fromEntries([...allowed].map((key) => [key, String(responses[key] || "")]));
  return {
    items,
    responses: semanticResponses,
    deterministicScore: Math.max(0, baseline.score - replaceScore),
    maxMarks: baseline.maxScore,
    unassessedMarks: baseline.unassessedMarks || 0,
    baselineScore: baseline.score
  };
}
